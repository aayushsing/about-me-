Smart Blind Stick – Details

Bill of Materials

  • Arduino Uno/Nano
  • Ultrasonic Sensor HC-SR04 (Trig/Echo)
  • 5V Active buzzer and/or vibration motor with NPN transistor + diode
  • Power: 5V power bank (recommended)
  • Jumper wires, light stick frame

Pin Map (Uno)

  • HC-SR04 Trig → D9
  • HC-SR04 Echo → D8
  • Buzzer → D5
  • Vibration Motor → D6 via NPN transistor (1k base, diode across motor)
  • 5V and GND → common rails

Logic

  1. Measure distance every 50 ms
  2. < 120 cm → slow beep
  3. < 80 cm → medium beep + weak vibration
  4. < 40 cm → rapid beep + strong vibration

Arduino Code (Uno/Nano) – Non-blocking beeps + PWM vibration 
// Smart Blind Stick – Ultrasonic (HC-SR04) + Buzzer + Vibration Motor
// Author: Aayush Singh (portfolio project)
// Board: Arduino Uno / Nano (ATmega328P)
//
// Wiring (Uno/Nano):
//   HC-SR04 Vcc  -> 5V
//   HC-SR04 GND  -> GND
//   HC-SR04 Trig -> D9
//   HC-SR04 Echo -> D8
//   Buzzer (+)   -> D5 (active buzzer; optional 220 Ω series resistor)
//   Buzzer (−)   -> GND
//   Vibration motor (+) -> +5V
//   Vibration motor (−) -> Collector of NPN (e.g., 2N2222/S8050)
//   NPN Emitter  -> GND
//   NPN Base     -> D6 via 1 kΩ resistor
//   Diode (1N4148/1N4007) across motor: Anode to NPN collector, Cathode to +5V
//
// Behavior:
//   - Measures distance ~20 times per second
//   - Three distance zones: FAR (<120 cm), MID (<80 cm), NEAR (<40 cm)
//   - Beep cadence and vibration strength increase as obstacles get closer
//   - Non-blocking (millis()), so tone() chirps are timed without delay()
//
// Safety:
//   - Ensure common ground for all modules
//   - Strain-relieve and secure wires along the stick
//   - Do not power motors directly from an I/O pin; always use a transistor

const uint8_t TRIG_PIN   = 9;
const uint8_t ECHO_PIN   = 8;
const uint8_t BUZZER_PIN = 5;   // uses tone()
const uint8_t VIB_PIN    = 6;   // PWM pin -> NPN base via 1k resistor

const uint16_t NEAR_TH_CM = 40;   // < 40 cm => rapid alerts
const uint16_t MID_TH_CM  = 80;   // < 80 cm => medium alerts
const uint16_t FAR_TH_CM  = 120;  // <120 cm => slow alerts
const uint16_t MAX_CM     = 300;  // ignore beyond this

const uint16_t BEEP_FREQ_HZ   = 2200; // buzzer tone
const uint8_t  VIB_NEAR_PWM   = 255;  // 0..255
const uint8_t  VIB_MID_PWM    = 160;
const uint8_t  VIB_FAR_PWM    = 80;
const uint8_t  VIB_OFF_PWM    = 0;

const uint16_t SAMPLE_INTERVAL_MS = 50;  // distance sample period
const uint16_t BEEP_NEAR_MS       = 120; // cadence for near
const uint16_t BEEP_MID_MS        = 250; // cadence for mid
const uint16_t BEEP_FAR_MS        = 450; // cadence for far
const uint8_t  BEEP_LEN_MS        = 50;  // chirp length

unsigned long lastSampleMs = 0;
unsigned long lastBeepMs   = 0;
int distanceCM             = 999;

int readDistanceCM() {
  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  unsigned long dur = pulseIn(ECHO_PIN, HIGH, 25000UL);
  if (dur == 0) return 999;
  float cm = dur / 58.0f;
  if (cm < 1 || cm > MAX_CM) return 999;
  return (int)cm;
}

int median3DistanceCM() {
  int a = readDistanceCM();
  int b = readDistanceCM();
  int c = readDistanceCM();
  if (a > b) { int t = a; a = b; b = t; }
  if (b > c) { int t = b; b = c; c = t; }
  if (a > b) { int t = a; a = b; b = t; }
  return b;
}

void setup() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(VIB_PIN, OUTPUT);
}

void loop() {
  unsigned long now = millis();

  if (now - lastSampleMs >= SAMPLE_INTERVAL_MS) {
    lastSampleMs = now;
    distanceCM = median3DistanceCM();
  }

  uint8_t vib = 0;
  if (distanceCM < NEAR_TH_CM)        vib = 255;
  else if (distanceCM < MID_TH_CM)    vib = 160;
  else if (distanceCM < FAR_TH_CM)    vib = 80;
  else                                vib = 0;
  analogWrite(VIB_PIN, vib);

  uint16_t beepInterval = 0;
  if (distanceCM < NEAR_TH_CM)        beepInterval = BEEP_NEAR_MS;
  else if (distanceCM < MID_TH_CM)    beepInterval = BEEP_MID_MS;
  else if (distanceCM < FAR_TH_CM)    beepInterval = BEEP_FAR_MS;

  if (beepInterval > 0 && (now - lastBeepMs >= beepInterval)) {
    lastBeepMs = now;
    tone(BUZZER_PIN, BEEP_FREQ_HZ, BEEP_LEN_MS);
  }
}
